Ex. No. 3: Functional Blockchain Prototype

Aim

Build a fully functional prototype of a blockchain system that supports transaction creation, transaction pooling (mempool), block generation, basic Proof of Work (PoW) consensus, and state management.

Procedure

  1. Create a project directory named FunctionalBlockchain.
  2. Implement a Transaction class representing transfer operations (sender, recipient, amount, timestamp) and string serialization.
  3. Implement a Mempool structure to temporarily hold pending transactions before block inclusion.
  4. Develop a Block class that packages transactions from the mempool, references the previous block's hash, and tracks a nonce value for Proof of Work.
  5. Implement a Proof of Work algorithm requiring block hashes to satisfy a configurable difficulty target (e.g., hash starting with a specific number of leading zeros).
  6. Construct a Blockchain class that manages the chain, validates incoming blocks, executes transactions to update state balances, and miner rewards.
  7. Add unit tests or execution scripts in if __name__ == "__main__": to submit transactions, mine blocks, check account balances, and verify chain validity.

Code

import time
import json
import hashlib
from typing import List, Dict

def sha256_hex(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()

class Transaction:
    def __init__(self, sender: str, recipient: str, amount: float):
        self.sender = sender
        self.recipient = recipient
        self.amount = amount
        self.timestamp = time.time()

    def to_dict( me ) -> Dict:
        return {
            "sender": self.sender,
            "recipient": self.recipient,
            "amount": self.amount,
            "timestamp": self.timestamp
        }

class Block:
    def __init__(self, index: int, transactions: List[Dict], previous_hash: str, nonce: int = 0):
        self.index = index
        self.timestamp = time.time()
        self.transactions = transactions
        self.previous_hash = previous_hash
        self.nonce = nonce
        self.hash = self.compute_hash()

    def compute_hash(self) -> str:
        block_string = json.dumps({
            "index": self.index,
            "timestamp": self.timestamp,
            "transactions": self.transactions,
            "previous_hash": self.previous_hash,
            "nonce": self.nonce
        }, sort_keys=True)
        return sha256_hex(block_string.encode())

    def mine_block(self, difficulty: int):
        target = "0" * difficulty
        while not self.hash.startswith(target):
            self.nonce += 1
            self.hash = self.compute_hash()

class Blockchain:
    def __init__(self, difficulty: int = 2):
        self.chain: List[Block] = []
        self.mempool: List[Dict] = []
        self.difficulty = difficulty
        self.balances: Dict[str, float] = {}
        self.create_genesis_block()

    def create_genesis_block(self):
        genesis = Block(0, [{"sender": "0", "recipient": "Genesis", "amount": 1000}], "0" * 64)
        genesis.mine_block(self.difficulty)
        self.chain.append(genesis)
        self.balances["Genesis"] = 1000

    def add_transaction(self, tx: Transaction) -> bool:
        sender_balance = self.balances.get(tx.sender, 0)
        if tx.sender != "0" and sender_balance < tx.amount:
            print(f"Transaction failed: Insufficient balance for {tx.sender}")
            return False
        self.mempool.append(tx.to_dict())
        return True

    def mine_pending_transactions(self, miner_address: str):
        # Add miner reward transaction
        reward_tx = Transaction(sender="0", recipient=miner_address, amount=10)
        self.mempool.append(reward_tx.to_dict())

        new_block = Block(
            index=len(self.chain),
            transactions=self.mempool.copy(),
            previous_hash=self.chain[-1].hash
        )
        new_block.mine_block(self.difficulty)
        
        # Apply transactions to state
        for tx in new_block.transactions:
            if tx["sender"] != "0":
                self.balances[tx["sender"]] -= tx["amount"]
            self.balances[tx["recipient"]] = self.balances.get(tx["recipient"], 0) + tx["amount"]

        self.chain.append(new_block)
        self.mempool = []

    def is_chain_valid(Self) -> bool:
        for i in range(1, len(self.chain)):
            curr = self.chain[i]
            prev = self.chain[i - 1]
            if curr.hash != curr.compute_hash():
                return False
            if curr.previous_hash != prev.hash:
                return False
        return True

if __name__ == "__main__":
    bc = Blockchain(difficulty=3)
    
    # Fund initial users
    bc.balances["Alice"] = 100
    bc.balances["Bob"] = 50

    bc.add_transaction(Transaction("Alice", "Bob", 20))
    bc.add_transaction(Transaction("Bob", "Charlie", 15))

    print("Mining block 1...")
    bc.mine_pending_transactions("Miner1")

    print("\nBalances:", bc.balances)
    print("Is chain valid?", bc.is_chain_valid())
