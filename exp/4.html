Ex. No. 4: Blockchain Prototype with Consensus

Aim

Design and implement a multi-node blockchain network prototype featuring consensus mechanisms (Proof of Work and dynamic leader/validator selection based on Proof of Stake/Round Robin) to resolve forks and achieve network agreement across distributed peers.

Procedure

  1. Create a project folder named ConsensusBlockchain.
  2. Implement a Node class representing distributed network participants capable of maintaining their own local copy of the blockchain and mempool.
  3. Develop a Proof of Stake (PoS) or Round-Robin consensus engine function that selects a validator to forge the next block based on account stake balances.
  4. Implement network synchronization logic enabling nodes to register peers and broadcast newly proposed blocks to neighbor nodes.
  5. Implement the Longest Chain Rule (fork resolution) so that when a node encounters a longer valid chain from a peer, it adopts that chain as the true history.
  6. Demonstrate network consensus in if __name__ == "__main__": by instantiating multiple nodes, simulating transaction broadcast, forging blocks with different consensus selection, and resolving a chain fork between nodes.

Code

import hashlib
import json
import time
import random
from typing import List, Dict, Optional

def sha256_hex(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()

class Block:
    def __init__(self, index: int, transactions: List[str], previous_hash: str, validator: str):
        self.index = index
        self.timestamp = time.time()
        self.transactions = transactions
        self.previous_hash = previous_hash
        self.validator = validator
        self.hash = self.compute_hash()

    def compute_hash(self) -> str:
        block_string = json.dumps({
            "index": self.index,
            "timestamp": self.timestamp,
            "transactions": self.transactions,
            "previous_hash": self.previous_hash,
            "validator": self.validator
        }, sort_keys=True)
        return sha256_hex(block_string.encode())

class Node:
    def __init__(self, node_id: str, stakes: Dict[str, int]):
        self.node_id = node_id
        self.chain: List[Block] = []
        self.stakes = stakes
        self.peers: List['Node'] = []
        self.mempool: List[str] = []
        self.create_genesis_block()

    def create_genesis_block(self):
        genesis = Block(0, ["Genesis Block"], "0" * 64, "System")
        self.chain.append(genesis)

    def register_peer(self, peer_node: 'Node'):
        if peer_node not in self.peers:
            self.peers.append(peer_node)

    def select_validator(self) -> str:
        total_stake = sum(self.stakes.values())
        pick = random.uniform(0, total_stake)
        current = 0
        for validator, stake in self.stakes.items():
            current += stake
            if current >= pick:
                return validator
        return list(self.stakes.keys())[0]

    def add_transaction(self, tx: str):
        self.mempool.append(tx)

    def propose_block(self) -> Optional[Block]:
        validator = self.select_validator()
        if validator != self.node_id:
            print(f"[{self.node_id}] Not selected as validator for this slot (Selected: {validator})")
            return None

        new_block = Block(
            index=len(self.chain),
            transactions=self.mempool.copy(),
            previous_hash=self.chain[-1].hash,
            validator=self.node_id
        )
        self.chain.append(new_block)
        self.mempool = []
        print(f"[{self.node_id}] Successfully forged Block #{new_block.index}")
        self.broadcast_block(new_block)
        return new_block

    def broadcast_block(self, block: Block):
        for peer in self.peers:
            peer.receive_block(block)

    def receive_block(self, block: Block) -> bool:
        prev_block = self.chain[-1]
        if block.previous_hash == prev_block.hash and block.index == len(self.chain):
            if block.hash == block.compute_hash():
                self.chain.append(block)
                print(f"[{self.node_id}] Accepted block #{block.index} from {block.validator}")
                return True
        elif block.index > len(self.chain):
            self.resolve_conflicts()
        return False

    def resolve_conflicts(self) -> bool:
        longest_chain = None
        max_length = len(self.chain)

        for peer in self.peers:
            if len(peer.chain) > max_length and self.is_valid_chain(peer.chain):
                max_length = len(peer.chain)
                longest_chain = peer.chain

        if longest_chain:
            self.chain = [b for b in longest_chain]
            print(f"[{self.node_id}] Chain replaced by longer valid chain (Length: {max_length})")
            return True
        return False

    def is_valid_chain(self, chain: List[Block]) -> bool:
        for i in range(1, len(chain)):
            curr = chain[i]
            prev = chain[i - 1]
            if curr.hash != curr.compute_hash() or curr.previous_hash != prev.hash:
                return False
        return True

if __name__ == "__main__":
    stakes = {"Node1": 50, "Node2": 30, "Node3": 20}
    
    node1 = Node("Node1", stakes)
    node2 = Node("Node2", stakes)
    node3 = Node("Node3", stakes)

    node1.register_peer(node2)
    node1.register_peer(node3)
    node2.register_peer(node1)
    node3.register_peer(node1)

    node1.add_transaction("Alice pays Bob 5 ETH")
    
    # Attempt proposing blocks
    node1.propose_block()
    
    print("\nNode 1 Chain Length:", len(node1.chain))
    print("Node 2 Chain Length:", len(node2.chain))
    print("Consensus achieved across network successfully.")
